  \int_a^b f'(x)\,dx = f(b) - f(a).

\subsection{Linear Algebra}
For an $n \times n$ matrix $A$ with eigenvalues $\lambda_1, \dots, \lambda_n$:
\begin{equation}
  \det(A) = \prod_{i=1}^{n} \lambda_i, \qquad \operatorname{tr}(A) = \sum_{i=1}^{n} \lambda_i.
\end{equation}

\begin{theorem}[Cauchy-Schwarz]
For vectors $\mathbf{u}, \mathbf{v} \in \mathbb{R}^n$:
\[
  |\langle \mathbf{u}, \mathbf{v} \rangle|^2 \le \langle \mathbf{u}, \mathbf{u} \rangle \cdot \langle \mathbf{v}, \mathbf{v} \rangle.
\]
\end{theorem}

\begin{lemma}
If $A$ is positive definite, then all eigenvalues are positive.
\end{lemma}
